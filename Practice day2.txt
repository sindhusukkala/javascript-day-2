 2. First 10 Prime Numbers
let count = 0;
let num = 2;
while (count < 10) {
  let isPrime = true;
  for (let i = 2; i <= Math.sqrt(num); i++) {
    if (num % i === 0) {
      isPrime = false;
      break;
    }
  }
  if (isPrime) {
    console.log("Prime:", num);
    count++;
  }
  num++;
}

 JavaScript Function: Multiplication Table Generator


function generateMultiplicationTable(n) {
  console.log(`Multiplication Table (${n} x ${n}):`);

  // Print column headers
  let header = "    ";
  for (let i = 1; i <= n; i++) {
    header += i.toString().padStart(4);
  }
  console.log(header);
  console.log("-".repeat(header.length));

  // Print table rows
  for (let i = 1; i <= n; i++) {
    let row = i.toString().padStart(3) + " |";
    for (let j = 1; j <= n; j++) {
      row += (i * j).toString().padStart(4);
    }
    console.log(row);
  }
}  

 Example Usage:
generateMultiplicationTable(5);
 Output:

Multiplication Table (5 x 5):
       1   2   3   4   5
-------------------------
  1 |   1   2   3   4   5
  2 |   2   4   6   8  10
  3 |   3   6   9  12  15
  4 |   4   8  12  16  20
  5 |   5  10  15  20  25
Explanation:
Outer loop (i) runs for rows.

Inner loop (j) runs for columns.

padStart() aligns values for neat formatting.

| and - improve table readability.
